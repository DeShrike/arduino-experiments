#pragma once
#include <Arduino.h>
#include <functional>

class Button {
public:
    Button(uint8_t pin, bool activeLow = true);

    void begin();
    void update();

    // Callback setters
    void onPress(std::function<void()> cb);
    void onDoublePress(std::function<void()> cb);
    void onLongPress(std::function<void()> cb);
    void onHoldRepeat(std::function<void()> cb);
    void onRelease(std::function<void()> cb);

    bool isDown() const;

private:
    uint8_t _pin;
    bool _activeLow;

    bool _state;
    bool _lastState;

    unsigned long _lastDebounceTime;
    unsigned long _pressTime;
    unsigned long _lastReleaseTime;
    unsigned long _lastHoldRepeatTime;

    bool _waitingForSecondPress;
    bool _longPressFired;

    std::function<void()> _onPress;
    std::function<void()> _onDoublePress;
    std::function<void()> _onLongPress;
    std::function<void()> _onHoldRepeat;
    std::function<void()> _onRelease;

    static constexpr unsigned long DEBOUNCE_MS      = 40;
    static constexpr unsigned long LONG_PRESS_MS   = 800;
    static constexpr unsigned long DOUBLE_PRESS_MS = 400;
    static constexpr unsigned long HOLD_REPEAT_MS  = 200;
};





#include "Button.h"

Button::Button(uint8_t pin, bool activeLow)
    : _pin(pin),
      _activeLow(activeLow),
      _state(false),
      _lastState(false),
      _lastDebounceTime(0),
      _pressTime(0),
      _lastReleaseTime(0),
      _lastHoldRepeatTime(0),
      _waitingForSecondPress(false),
      _longPressFired(false) {}

void Button::begin() {
    pinMode(_pin, _activeLow ? INPUT_PULLUP : INPUT);
}

bool Button::isDown() const {
    return _state;
}

void Button::onPress(std::function<void()> cb)       { _onPress = cb; }
void Button::onDoublePress(std::function<void()> cb) { _onDoublePress = cb; }
void Button::onLongPress(std::function<void()> cb)   { _onLongPress = cb; }
void Button::onHoldRepeat(std::function<void()> cb)  { _onHoldRepeat = cb; }
void Button::onRelease(std::function<void()> cb)     { _onRelease = cb; }

void Button::update() {
    unsigned long now = millis();

    bool reading = digitalRead(_pin);
    bool pressed = _activeLow ? !reading : reading;

    // Debounce
    if (pressed != _lastState) {
        _lastDebounceTime = now;
    }

    if ((now - _lastDebounceTime) < DEBOUNCE_MS) {
        _lastState = pressed;
        return;
    }

    _lastState = pressed;

    // Just pressed
    if (pressed && !_state) {
        _state = true;
        _pressTime = now;
        _lastHoldRepeatTime = now;
        _longPressFired = false;
        return;
    }

    // Held
    if (pressed && _state) {
        if (!_longPressFired && (now - _pressTime >= LONG_PRESS_MS)) {
            _longPressFired = true;
            _lastHoldRepeatTime = now;
            if (_onLongPress) _onLongPress();
            return;
        }

        if (_longPressFired &&
            (now - _lastHoldRepeatTime >= HOLD_REPEAT_MS)) {
            _lastHoldRepeatTime = now;
            if (_onHoldRepeat) _onHoldRepeat();
            return;
        }
        return;
    }

    // Released
    if (!pressed && _state) {
        _state = false;

        if (_onRelease) _onRelease();

        if (_longPressFired) {
            _waitingForSecondPress = false;
            return;
        }

        if (_waitingForSecondPress &&
            (now - _lastReleaseTime <= DOUBLE_PRESS_MS)) {
            _waitingForSecondPress = false;
            if (_onDoublePress) _onDoublePress();
            return;
        }

        _waitingForSecondPress = true;
        _lastReleaseTime = now;
        return;
    }

    // Single press timeout
    if (_waitingForSecondPress &&
        (now - _lastReleaseTime > DOUBLE_PRESS_MS)) {
        _waitingForSecondPress = false;
        if (_onPress) _onPress();
    }
}











#include "Button.h"

Button btn(0);

void setup() {
    Serial.begin(115200);
    btn.begin();

    btn.onPress([]() {
        Serial.println("Pressed");
    });

    btn.onDoublePress([]() {
        Serial.println("Double press");
    });

    btn.onLongPress([]() {
        Serial.println("Long press");
    });

    btn.onHoldRepeat([]() {
        Serial.println("Repeating...");
    });

    btn.onRelease([]() {
        Serial.println("Released");
    });
}

void loop() {
    btn.update();
}
